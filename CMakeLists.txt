cmake_minimum_required(VERSION 3.20)
project(badapple_generator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(badapple_geometry generator/cpp/geometry.cpp)
if(WIN32)
 add_executable(badapple_extract generator/cpp/extract.cpp)
 target_link_libraries(badapple_extract PRIVATE mfreadwrite mfplat mfuuid ole32 propsys)
 if(MINGW)
  target_link_options(badapple_extract PRIVATE -municode)
 endif()
else()
 message(FATAL_ERROR "The fixed reference-video extraction path requires Windows Media Foundation")
endif()
foreach(target badapple_geometry badapple_extract)
 if(MSVC)
  target_compile_options(${target} PRIVATE /O2 /fp:precise)
 else()
  target_compile_options(${target} PRIVATE -O2 -fno-fast-math -ffp-contract=off)
  if(MINGW)
   target_link_options(${target} PRIVATE -static)
  endif()
 endif()
endforeach()
